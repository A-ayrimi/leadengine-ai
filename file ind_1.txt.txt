//@version=6
indicator("Pivot Signals - 3 TF (No Repaint)", overlay=true, max_labels_count=300)

f_labelSize(v) =>
    switch v
        1 => size.tiny
        2 => size.small
        3 => size.normal
        4 => size.large
        => size.huge

// پیوت‌ها و زمان واقعی بار پیوت (در گذشته) از تایم‌فریم مورد نظر
f_pivotData(L, R) =>
    ph = ta.pivothigh(high, L, R)
    pl = ta.pivotlow(low, L, R)
    [ph, pl, na(ph) ? na : time[R], na(pl) ? na : time[R]]

f_instance(tf, L, R, showLabels, buyCol, sellCol, sz, tag) =>
    [ph, pl, phT, plT] = request.security(syminfo.tickerid, tf, f_pivotData(L, R),
         barmerge.gaps_off, barmerge.lookahead_off)

    // روی تایم‌فریم بالاتر، مقدار پیوت تا پایان آن بار تکرار می‌شود؛ فقط اولین بار را می‌گیریم
    newHigh = not na(phT) and na(phT[1]) and barstate.isconfirmed
    newLow  = not na(plT) and na(plT[1]) and barstate.isconfirmed

    if newHigh
        if showLabels
            label.new(phT, ph, tag + " Sell", xloc.bar_time, yloc.price,
                 color.new(sellCol, 0), label.style_label_down, color.white, sz)
        alert(tag + " Sell signal (confirmed pivot high)", alert.freq_once_per_bar_close)

    if newLow
        if showLabels
            label.new(plT, pl, tag + " Buy", xloc.bar_time, yloc.price,
                 color.new(buyCol, 0), label.style_label_up, color.white, sz)
        alert(tag + " Buy signal (confirmed pivot low)", alert.freq_once_per_bar_close)

// ---------- Instance 1 ----------

buy1  = input.color(#03ff85, "Buy color", group=g1)
sell1 = input.color(#fc0808, "Sell color", group=g1)
sz1   = f_labelSize(input.int(3, "Label size", minval=1, maxval=5, group=g1))
f_instance(tf1, L1, R1, show1, buy1, sell1, sz1, "I1")
