// ───────────────────────── ① Kalman Filter ─────────────────────────
gK       = "① Kalman Filter"
src      = input.source(hl2,  "Source", group = gK)
kQ       = input.float(0.05,  "Process Noise Q  (↑ = واکنش سریع‌تر)", minval = 0.0001, step = 0.005, group = gK)
kR       = input.float(1.0,   "Measurement Noise R  (↑ = نرم‌تر)",    minval = 0.01,   step = 0.1,   group = gK)
kLead    = input.float(0.8,   "Lag Compensation  (جبران تأخیر 0..3)", minval = 0.0, maxval = 3.0, step = 0.1, group = gK)
adaptive = input.bool(true,   "Adaptive Q  (شتاب‌گیری خودکار در شکست‌ها)", group = gK)

// ─────────────────────── ② Adaptive Supertrend ─────────────────────
gS      = "② Adaptive Supertrend"
atrLen  = input.int(10,   "ATR Length", minval = 1, group = gS)
stMult  = input.float(1.8,"ATR Multiplier", minval = 0.25, step = 0.1, group = gS)
bufMult = input.float(0.05,"Breakout Buffer × ATR (ضد نویز)", minval = 0, step = 0.05, group = gS)
sigMode = input.string("Bar Close (No Repaint)", "Signal Timing",
             options = ["Bar Close (No Repaint)", "Real-Time (Intrabar)"], group = gS)

// ──────────────────────── ③ VWMA OB/OS Zones ───────────────────────
gV       = "③ VWMA OB/OS Zones"
showZone = input.bool(true, "نمایش زون‌ها", group = gV)
vLen     = input.int(20,    "VWMA / HH-LL Length", minval = 2, group = gV)
zMultU   = input.float(1.0005, "Upper Multiplier", step = 0.0005, group = gV)
zMultL   = input.float(0.9995, "Lower Multiplier", step = 0.0005, group = gV)
zWidth   = input.float(1.2, "Zone Width × ATR", minval = 0.1, step = 0.1, group = gV)

// ──────────────────── ④ Higher-TF Confluence (اختیاری) ─────────────
gH       = "④ Higher-TF Filter"
useHTF   = input.bool(false, "فیلتر تایم‌فریم بالاتر (تأخیر اضافه می‌کند)", group = gH)
htfTF    = input.timeframe("15", "HTF", group = gH)
htfSoft  = input.bool(true,  "Soft mode (آخرین وضعیت زنده HTF = کم‌تأخیر)", group = gH)
showPanel= input.bool(true,  "نمایش جدول وضعیت", group = gH)

// ─────────────────────── ⑤ Reversal / TP / Visual ──────────────────
gR       = "⑤ Reversal & TP"
showRev  = input.bool(true, "برچسب R + خطوط عمودی ریورسال", group = gR)
showTP   = input.bool(true, "نقاط Take-Profit (X)", group = gR)
maxTP    = input.int(2, "حداکثر TP در هر معامله", minval = 1, maxval = 5, group = gR)
buyCol   = input.color(#00e676, "Buy Color",  group = gR)
sellCol  = input.color(#ff1744, "Sell Color", group = gR)

// ════════════════════ Kalman 2-State (position + velocity) ═════════
// x(k) = x(k-1) + v(k-1) ;  v(k) = v(k-1) ;  z = x + noise
f_kalman(float z, float q, float r, float lead) =>
    var float x   = na
    var float v   = 0.0
    var float p00 = 1.0, p01 = 0.0, p10 = 0.0, p11 = 1.0
    float out = na
    if na(x)
        x   := z
        out := z
    else
        // ── Predict
        float xp  = x + v
        float vp  = v
        float a00 = p00 + p01 + p10 + p11 + q
        float a01 = p01 + p11
        float a10 = p10 + p11
        float a11 = p11 + q
        // ── Update (Kalman Gain)
        float s   = a00 + r
        float k0  = a00 / s
        float k1  = a10 / s
        float y   = z - xp
        x   := xp + k0 * y
        v   := vp + k1 * y
        p00 := a00 - k0 * a00
        p01 := a01 - k0 * a01
        p10 := a10 - k1 * a00
        p11 := a11 - k1 * a01
        // خروجی + جبران تأخیر با بردار سرعت  ← کلید "Zero-Lag"
        out := x + lead * v
    out

// ════════════════════ هسته: Kalman-Adapted Supertrend ══════════════
f_core() =>
    float aFast = ta.atr(14)
    float aSlow = ta.atr(50)
    float ratio = na(aSlow) or aSlow == 0 ? 1.0 : aFast / aSlow
    // Q تطبیقی: در انفجار نوسان فیلتر سریع می‌شود → سیگنال در لحظهٔ شکست
    float qEff  = adaptive ? kQ * math.pow(math.max(0.3, math.min(3.0, ratio)), 2) : kQ

    float kf   = f_kalman(src, qEff, kR, kLead)
    float atrv = ta.atr(atrLen)
    float up   = kf - stMult * atrv
    float dn   = kf + stMult * atrv
    float buf  = bufMult * atrv

    var float ls = na
    var float ss = na
    var int   d  = 1

    float pLs = nz(ls, up)
    float pSs = nz(ss, dn)
    ls := close[1] > pLs ? math.max(up, pLs) : up      // trailing stop صعودی
    ss := close[1] < pSs ? math.min(dn, pSs) : dn      // trailing stop نزولی
    d  := close > pSs + buf ? 1 : close < pLs - buf ? -1 : d

    float st = d == 1 ? ls : ss
    [d, st, kf, atrv]

f_dirOnly() =>
    [d, _s, _k, _a] = f_core()
    htfSoft ? d : d[1]          // غیرسافت = فقط کندل بسته‌شدهٔ HTF (ضد ریپینت کامل)

[dir, stLine, kf, atrv] = f_core()

int htfDir = 0
if useHTF
    htfDir := request.security(syminfo.tickerid, htfTF, f_dirOnly(),
                 barmerge.gaps_off, barmerge.lookahead_off)

// ════════════════════════ VWMA OB/OS Zones ═════════════════════════
float u1 = ta.vwma(ta.highest(high, vLen), vLen) * zMultU
float l1 = ta.vwma(ta.lowest(low,  vLen), vLen) * zMultL
float u2 = u1 + zWidth * atrv
float l2 = l1 - zWidth * atrv

// ══════════════════════════ SIGNALS ════════════════════════════════
bool confirmed = sigMode == "Bar Close (No Repaint)" ? barstate.isconfirmed : true
bool flipUp    = dir ==  1 and dir[1] == -1
bool flipDn    = dir == -1 and dir[1] ==  1
bool okLong    = not useHTF or htfDir >=  0
bool okShort   = not useHTF or htfDir <=  0

bool buySig  = flipUp and okLong  and confirmed
bool sellSig = flipDn and okShort and confirmed

// ═══════════════════ Trade State / TP / Entry Table ════════════════
var int   pos     = 0
var float entryP  = na
var int   entryB  = na
var int   tpCount = 0

if buySig
    pos := 1,  entryP := close, entryB := bar_index, tpCount := 0
if sellSig
    pos := -1, entryP := close, entryB := bar_index, tpCount := 0

// TP ← ورود به زون OB/OS + ساختار HH/LL شکست‌خورده
bool tpLong  = showTP and pos ==  1 and tpCount < maxTP and high >= u1 and high < high[1] and close < close[1]
bool tpShort = showTP and pos == -1 and tpCount < maxTP and low  <= l1 and low  > low[1]  and close > close[1]
if (tpLong or tpShort) and confirmed
    tpCount += 1

// Reversal (R) ← رد شدن قیمت از زون
bool revDown = showRev and ta.crossunder(close, u1) and close < open and confirmed
bool revUp   = showRev and ta.crossover(close,  l1) and close > open and confirmed

// ════════════════════════════ PLOTS ════════════════════════════════
pST = plot(stLine, "Kalman Supertrend", dir == 1 ? buyCol : sellCol, 2)
plot(kf, "Kalman Line", color.new(color.orange, 35), 1)

pU1 = plot(showZone ? u1 : na, "OB inner", color.new(sellCol, 70))
pU2 = plot(showZone ? u2 : na, "OB outer", color.new(sellCol, 70))
pL1 = plot(showZone ? l1 : na, "OS inner", color.new(buyCol, 70))
pL2 = plot(showZone ? l2 : na, "OS outer", color.new(buyCol, 70))
fill(pU1, pU2, color.new(sellCol, 85), "OB Zone")
fill(pL1, pL2, color.new(buyCol,  85), "OS Zone")

plotshape(buySig,  "BUY",  shape.labelup,   location.belowbar, buyCol,  text = "BUY",
          textcolor = color.white, size = size.small)
plotshape(sellSig, "SELL", shape.labeldown, location.abovebar, sellCol, text = "SELL",
          textcolor = color.white, size = size.small)

plotshape(tpLong and confirmed,  "TP Long",  shape.xcross, location.abovebar,
          color.new(color.lime, 0), size = size.tiny)
plotshape(tpShort and confirmed, "TP Short", shape.xcross, location.belowbar,
          color.new(color.red, 0),  size = size.tiny)

plotshape(revDown, "R Down", shape.labeldown, location.abovebar,
          color.new(sellCol, 25), text = "R", textcolor = color.white, size = size.tiny)
plotshape(revUp,   "R Up",   shape.labelup,   location.belowbar,
          color.new(buyCol, 25),  text = "R", textcolor = color.white, size = size.tiny)

bgcolor(revDown ? color.new(sellCol, 82) : revUp ? color.new(buyCol, 82) : na)

// ═════════════════════════ Status Table ════════════════════════════

    float pnl = na(entryP) ? na : (pos == 1 ? close - entryP : entryP - close)
    table.cell(t, 0, 0, "KST Pro", bgcolor = color.new(color.gray, 60), text_color = color.white)
    table.cell(t, 1, 0, timeframe.period,  bgcolor = color.new(color.gray, 60), text_color = color.white)
    table.cell(t, 0, 1, "Trend", text_color = color.white)
    table.cell(t, 1, 1, dir == 1 ? "LONG ▲" : "SHORT ▼",
               text_color = dir == 1 ? buyCol : sellCol)
    table.cell(t, 0, 2, "Entry", text_color = color.white)
    table.cell(t, 1, 2, na(entryP) ? "—" : str.tostring(entryP, format.mintick), text_color = color.white)
    table.cell(t, 0, 3, "Open P/L", text_color = color.white)
    table.cell(t, 1, 3, na(pnl) ? "—" : str.tostring(pnl, format.mintick),
               text_color = nz(pnl) >= 0 ? buyCol : sellCol)
    table.cell(t, 0, 4, "HTF " + (useHTF ? htfTF : "off"), text_color = color.white)
    table.cell(t, 1, 4, not useHTF ? "—" : htfDir == 1 ? "▲" : "▼",
               text_color = htfDir == 1 ? buyCol : sellCol)
